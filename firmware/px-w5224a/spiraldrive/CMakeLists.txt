cmake_minimum_required(VERSION "3.21")
project("spiraldrive_pxw5224a" LANGUAGES NONE)

set(KEIL_C251_ROOT "C:/Keil_v5/C251" CACHE PATH "Keil C251 installation directory")
include("cmake/KeilC251.cmake")

get_filename_component(SPIRALDRIVE_REPOSITORY_ROOT "../../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(SPIRALDRIVE_ORIGINAL_FIRMWARE "${SPIRALDRIVE_REPOSITORY_ROOT}/data/px-w5224a/MONT_104.bin" CACHE FILEPATH "Official PX-W5224A 1.04 MONT_104.bin")
set(SPIRALDRIVE_PATCH_DESCRIPTION "${CMAKE_CURRENT_SOURCE_DIR}/../mont-104.json")
set(SPIRALDRIVE_PATCHER_ROOT "${SPIRALDRIVE_REPOSITORY_ROOT}/tools/fwpatch-plextor")
find_program(CARGO "cargo" REQUIRED)
if(CMAKE_HOST_WIN32)
  set(SPIRALDRIVE_PATCHER "${SPIRALDRIVE_PATCHER_ROOT}/target/release/fwpatch-plextor.exe")
else()
  set(SPIRALDRIVE_PATCHER "${SPIRALDRIVE_PATCHER_ROOT}/target/release/fwpatch-plextor")
endif()
add_custom_target("fwpatch-plextor"
  COMMAND "${CARGO}" "build" "--release" "--locked" "--manifest-path" "${SPIRALDRIVE_PATCHER_ROOT}/Cargo.toml" "--target-dir" "${SPIRALDRIVE_PATCHER_ROOT}/target"
  BYPRODUCTS "${SPIRALDRIVE_PATCHER}" VERBATIM)

get_filename_component(SPIRALDRIVE_FIRMWARE_STEM "${SPIRALDRIVE_ORIGINAL_FIRMWARE}" NAME_WLE)
get_filename_component(SPIRALDRIVE_FIRMWARE_EXTENSION "${SPIRALDRIVE_ORIGINAL_FIRMWARE}" LAST_EXT)
set(SPIRALDRIVE_PATCHED_FIRMWARE "${CMAKE_CURRENT_BINARY_DIR}/${SPIRALDRIVE_FIRMWARE_STEM}_patched${SPIRALDRIVE_FIRMWARE_EXTENSION}")
add_custom_command(OUTPUT "${SPIRALDRIVE_PATCHED_FIRMWARE}"
  COMMAND "${SPIRALDRIVE_PATCHER}" "${SPIRALDRIVE_ORIGINAL_FIRMWARE}" "${SPIRALDRIVE_PATCH_DESCRIPTION}"
  DEPENDS "payload" "fwpatch-plextor" "${SPIRALDRIVE_PATCHER}" "${SPIRALDRIVE_ORIGINAL_FIRMWARE}" "${SPIRALDRIVE_PATCH_DESCRIPTION}"
    "$<TARGET_PROPERTY:payload,KEIL_C251_HEX_FILE>" "$<TARGET_PROPERTY:payload,KEIL_C251_MAP_FILE>"
  WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}" VERBATIM)
add_custom_target("firmware" ALL DEPENDS "${SPIRALDRIVE_PATCHED_FIRMWARE}")

find_program(CLANG_FORMAT "clang-format")
if(CLANG_FORMAT)
  file(GLOB_RECURSE SPIRALDRIVE_FORMAT_SOURCES CONFIGURE_DEPENDS "src/*.c" "src/*.h")
  add_custom_target("format" COMMAND "${CLANG_FORMAT}" "-i" ${SPIRALDRIVE_FORMAT_SOURCES} VERBATIM)
  add_custom_target("check-format" COMMAND "${CLANG_FORMAT}" "--dry-run" "--Werror" ${SPIRALDRIVE_FORMAT_SOURCES} VERBATIM)
endif()

add_subdirectory("src")
